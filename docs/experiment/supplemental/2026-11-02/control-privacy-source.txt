15: function isGregorianDate(value: string): boolean {
16:   if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
17:
18:   const [year, month, day] = value.split("-").map(Number);
19:   if (year < 1 || month < 1 || month > 12 || day < 1) return false;
20:
21:   const leapYear = year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
22:   const daysInMonth = [31, leapYear ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
23:   return day <= daysInMonth[month - 1];
24: }
25:
26: function isTimeZone(value: string): boolean {
27:   // Require a named zone rather than an abbreviation or fixed UTC offset.
28:   if (value !== "UTC" && !/^[A-Za-z_+-]+(?:\/[A-Za-z0-9_+-]+)+$/.test(value)) {
29:     return false;
30:   }
31:   try {
32:     new Intl.DateTimeFormat("en", { timeZone: value });
33:     return true;
34:   } catch {
35:     return false;
36:   }
37: }
38:
39: /** Checks syntax only; does not resolve a local time to an instant or log inputs. */
40: export function validateBirthInput(input: BirthInputValues): BirthInputValidation {
41:   const values = {
42:     date: input.date.trim(),
43:     time: input.time.trim(),
44:     timeZone: input.timeZone.trim(),
45:   };
46:   const errors: BirthInputErrors = {};
47:
48:   if (!values.date) {
49:     errors.date = "Enter a birth date in YYYY-MM-DD format.";
50:   } else if (!isGregorianDate(values.date)) {
51:     errors.date = "Enter a real Gregorian date in YYYY-MM-DD format (years 0001–9999).";
52:   }
53:
54:   if (!values.time) {
55:     errors.time = "Enter a birth time in 24-hour HH:MM format.";
56:   } else if (!/^([01]\d|2[0-3]):[0-5]\d$/.test(values.time)) {
57:     errors.time = "Enter a time from 00:00 to 23:59 in 24-hour HH:MM format.";
58:   }
59:
60:   if (!values.timeZone) {
61:     errors.timeZone = "Enter the birth location’s time zone, such as America/Los_Angeles or UTC.";
62:   } else if (!isTimeZone(values.timeZone)) {
63:     errors.timeZone = "Enter a recognized IANA time zone, such as Asia/Shanghai or America/Los_Angeles, or UTC; avoid abbreviations such as PST.";
64:   }
65:
66:   return Object.keys(errors).length ? { valid: false, errors } : { valid: true, values };
67: }
