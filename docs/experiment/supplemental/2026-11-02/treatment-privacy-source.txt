11: export type BirthInputValidationResult =
12:   | { valid: true; value: BirthInput }
13:   | { valid: false; errors: BirthInputErrors };

15: function isGregorianDate(value: string): boolean {
16:   if (value.length !== 10 || !/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
17:
18:   const [year, month, day] = value.split("-").map(Number);
19:   if (year < 1 || month < 1 || month > 12 || day < 1) return false;
20:
21:   const leapYear = year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
22:   const daysInMonth = [31, leapYear ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
23:   return day <= daysInMonth[month - 1];
24: }
25:
26: function isSupportedTimeZone(value: string): boolean {
27:   // Require a named zone, rather than a potentially ambiguous abbreviation or offset.
28:   if (value !== "UTC" && !/^[A-Za-z_+-]+(?:\/[A-Za-z0-9_+-]+)+$/.test(value)) {
29:     return false;
30:   }
31:
32:   try {
33:     new Intl.DateTimeFormat("en", { timeZone: value });
34:     return true;
35:   } catch {
36:     return false;
37:   }
38: }
39:
40: export function validateBirthInput(input: unknown): BirthInputValidationResult {
41:   const fields =
42:     typeof input === "object" && input !== null && !Array.isArray(input) ? input : {};
43:   const birthDate = "birthDate" in fields ? fields.birthDate : undefined;
44:   const birthTime = "birthTime" in fields ? fields.birthTime : undefined;
45:   const timeZone = "timeZone" in fields ? fields.timeZone : undefined;
46:   const errors: BirthInputErrors = {};
47:
48:   if (typeof birthDate !== "string" || birthDate === "") {
49:     errors.birthDate = "Enter a birth date in YYYY-MM-DD format.";
50:   } else if (!isGregorianDate(birthDate)) {
51:     errors.birthDate = "Enter a real Gregorian date in YYYY-MM-DD format (years 0001–9999).";
52:   }
53:
54:   if (typeof birthTime !== "string" || birthTime === "") {
55:     errors.birthTime = "Enter the local birth time in HH:mm format.";
56:   } else if (birthTime.length !== 5 || !/^([01]\d|2[0-3]):[0-5]\d$/.test(birthTime)) {
57:     errors.birthTime = "Enter a 24-hour time from 00:00 to 23:59, for example 14:30.";
58:   }
59:
60:   if (typeof timeZone !== "string" || timeZone === "") {
61:     errors.timeZone = "Enter the birth location’s IANA timezone, for example Asia/Taipei, or UTC.";
62:   } else if (!isSupportedTimeZone(timeZone)) {
63:     errors.timeZone = "Enter a timezone supported by this browser, such as Asia/Taipei or America/Los_Angeles, or UTC. Do not use an abbreviation or numeric offset.";
64:   }
65:
66:   if (
67:     Object.keys(errors).length > 0 ||
68:     typeof birthDate !== "string" ||
69:     typeof birthTime !== "string" ||
70:     typeof timeZone !== "string"
71:   ) {
72:     return { valid: false, errors };
73:   }
74:
75:   return { valid: true, value: { birthDate, birthTime, timeZone } };
76: }
